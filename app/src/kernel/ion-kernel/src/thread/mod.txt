// ================================
// Thread states and identifiers
// ================================

pub type ThreadId = u64;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ThreadState {
    Ready,
    Running,
    Blocked(BlockReason),
    Terminated,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum BlockReason {
    Join { child: ThreadId },
}

// ================================
// Joinable results
// ================================

pub struct ThreadResult<T> {
    ready: AtomicBool,
    value: UnsafeCell<Option<T>>,
}

unsafe impl<T: Send> Send for ThreadResult<T> {}
unsafe impl<T: Send> Sync for ThreadResult<T> {}

impl<T> ThreadResult<T> {
    fn new() -> Self {
        Self {
            ready: AtomicBool::new(false),
            value: UnsafeCell::new(None),
        }
    }
    fn set(&self, v: T) {
        unsafe { *self.value.get() = Some(v); }
        self.ready.store(true, Ordering::Release);
    }
    fn take(&self) -> T {
        unsafe { (*self.value.get()).take().expect("result not set") }
    }
}

// ================================
// Thread representation (TCB)
// ================================

pub struct Thread {
    pub id: ThreadId,
    pub state: ThreadState,

    // Owned stack and saved stack pointer (top-of-stack, grows down).
    _stack_mem: Box<[u8]>,
    pub stack_ptr: *mut u8,

    // Joiners waiting for this thread to finish (thread IDs).
    waiters: Vec<ThreadId>,

    // Optional join result slot. None for detached threads.
    join_raw: Option<NonNullErasedResult>,
}

#[derive(Clone, Copy)]
pub struct NonNullErasedResult {
    ptr: *const (),
}
impl NonNullErasedResult {
    pub fn new<T>(r: &ThreadResult<T>) -> Self {
        Self { ptr: r as *const ThreadResult<T> as *const () }
    }
    /// Reference to the result.
    /// # Safety
    /// ptr must be valid
    pub unsafe fn as_ref<T>(&self) -> &ThreadResult<T> {
        unsafe { &*(self.ptr as *const ThreadResult<T>) }
    }
}

impl Thread {
    pub fn new(id: ThreadId, stack_size: usize, entry: extern "C" fn(*mut ()) -> !) -> Self {
        let mut stack_mem = vec![0u8; stack_size].into_boxed_slice();
        let base = stack_mem.as_mut_ptr();
        let top = unsafe { base.add(stack_size) };

        // Align down to 16-byte boundary
        let aligned_top = (top as usize & !0xF) as *mut u8;
        serial_println!("initializing stack frame");
        let sp = unsafe {
            init_stack_frame(
                aligned_top,
                entry,
                SegmentSelector::new(1, x86_64::PrivilegeLevel::Ring0),
                RFlags::INTERRUPT_FLAG,
            )
        };

        Self {
            id,
            state: ThreadState::Ready,
            _stack_mem: stack_mem,
            stack_ptr: sp,
            waiters: Vec::new(),
            join_raw: None,
        }
    }
}

// ================================
// Initial stack frame and entry trampoline
// ================================

#[allow(unsafe_op_in_unsafe_fn)]
unsafe fn init_stack_frame(
    mut top: *mut u8,
    start_fn: extern "C" fn(*mut ()) -> !,
    cs: SegmentSelector,
    rflags: RFlags,
) -> *mut u8 {
    use core::mem::size_of;
    let q = size_of::<u64>();

    // Align down to 16 bytes so the RIP slot we return will be 16n.
    top = ((top as usize) & !0xF) as *mut u8;

    // Place callee-saved placeholders (r15, r14, r13, r12, rbx, rbp)
    // These live at the high end of the stack (above start_fn/ctx).
    for _ in 0..6 {
        top = top.sub(q);
        *(top as *mut u64) = 0xDEADBEEFDEADBEEF;
    }

    // We must write the operands so that memory (increasing addresses)
    // becomes: [RIP][CS][RFLAGS][PAD][ctx][start_fn][callee-saved...]
    //
    // To achieve that, write start_fn first (highest of the two), then ctx,
    // then PAD, then the interrupt frame (RFLAGS, CS, RIP) so that the final
    // returned pointer points at RIP.
    top = top.sub(q); // start_fn (will be at stack_ptr + 40)
    *(top as *mut u64) = start_fn as usize as u64;

    top = top.sub(q); // ctx (will be at stack_ptr + 32); fill later in spawn
    *(top as *mut u64) = 0;

    top = top.sub(q); // PAD (fixed 8 bytes at stack_ptr + 24)
    *(top as *mut u64) = 0u64;

    // Interrupt frame: RFLAGS, CS, RIP (write in reverse so returned pointer points at RIP)
    top = top.sub(q); // RFLAGS (stack_ptr + 16)
    *(top as *mut u64) = rflags.bits();
    top = top.sub(q); // CS (stack_ptr + 8)
    *(top as *mut u64) = cs.0 as u64;
    top = top.sub(q); // RIP (stack_ptr + 0)
    *(top as *mut u64) = thread_entry_stub_raw as *const () as u64;

    // Ensure alignment invariant: returned pointer (RIP slot) must be 16n
    debug_assert_eq!((top as usize) & 0xF, 0, "RSP must be 16n at iretq entry");

    top
}

/// # Safety
/// arguments must be placed correctly
#[unsafe(naked)]
pub unsafe extern "C" fn thread_entry_stub_raw() {
    core::arch::naked_asm!(
        r#"
        // Stack on entry (after iretq):
        //   rsp -> [RIP popped by iretq already]
        //          [CS popped]
        //          [RFLAGS popped]
        //          [PAD]
        //          [start_fn_ptr]
        //          [ctx_ptr]
        //          [callee-saved placeholders...]
        //
        // But our stub expects ctx then start_fn on the stack (pop rdi; pop rax).
        pop rdi          // rdi = ctx_ptr
        pop rax          // rax = start_fn_ptr
        jmp rax          // tail-call start_fn(ctx_ptr)
        "#
    );
}

#[inline(always)]
pub unsafe fn thread_entry_stub() -> ! {
    unsafe {
        thread_entry_stub_raw();
        core::hint::unreachable_unchecked()
    }
}

// ================================
// iretq-based context switch primitive
// ================================

#[unsafe(naked)]
pub unsafe extern "C" fn switch_interrupt_context_raw(
    old_sp: *mut *mut u8,
    new_sp: *const *mut u8,
) {
    core::arch::naked_asm!(
        r#"
        // Save current stack pointer (points at whatever the interrupted frame is)
        mov [rdi], rsp
        // Load next thread's interrupt frame (rsp should point at RIP slot)
        mov rsp, [rsi]
        // Return into next thread's stub (pops RIP, CS, RFLAGS)
        iretq
        "#,
    );
}

#[allow(unsafe_op_in_unsafe_fn)]
// new_sp_ptr: *const *mut u8  (address of the TCB.stack_ptr for next thread)
unsafe fn dump_and_assert_next_frame(new_sp_ptr: *const *mut u8) {
    let rsp = *new_sp_ptr as *const u64;

    // 1) Alignment
    debug_assert_eq!((rsp as usize) & 0xF, 0, "iretq entry must be 16-byte aligned");

    // 2) Dump the three words iretq will pop
    let rip = *rsp.add(0);
    let cs  = *rsp.add(1);
    let rf  = *rsp.add(2);
    serial_println!("NEXT FRAME: RIP={:#018x} CS={:#06x} RFLAGS={:#018x}", rip, cs as u16, rf);

    // 3) Dump the words after the frame (ctx, start_fn, placeholders)
    let ctx   = *rsp.add(3);
    let start = *rsp.add(4);
    serial_println!("AFTER FRAME: ctx={:#018x} start={:#018x}", ctx, start);
    
    // 4) Quick sanity checks
    debug_assert!(rip != 0 && rip != 0x4444_4444_4444_4444 && rip != 0xDEADBEEFDEADBEEF,
        "RIP looks like garbage; stack_ptr likely wrong or stack corrupted");
    debug_assert_eq!(cs as u16 & 0xFFF8, 1 << 3, "CS selector unexpected");
}

// Keep the old ret-based switcher around if you want to use it for special cooperative stacks.
// Do not call it on iretq-style stacks.
#[allow(dead_code)]
#[unsafe(naked)]
pub unsafe extern "C" fn switch_context_raw(old_sp: *mut *mut u8, new_sp: *const *mut u8) {
    core::arch::naked_asm!(
        r#"
        push rbp
        push rbx
        push r12
        push r13
        push r14
        push r15

        mov [rdi], rsp
        mov rsp, [rsi]

        pop r15
        pop r14
        pop r13
        pop r12
        pop rbx
        pop rbp

        ret
        "#
    );
}

#[inline(always)]
pub unsafe fn switch_context(_old_sp: *mut *mut u8, _new_sp: *const *mut u8) -> ! {
    // Not used for iretq stacks; kept for completeness.
    unsafe { core::hint::unreachable_unchecked() }
}

// ================================
// JoinHandle<T> API
// ================================

pub struct JoinHandle<T> {
    child: ThreadId,
    result: Box<ThreadResult<T>>,
}

impl<T: Send + 'static> JoinHandle<T> {
    #[allow(static_mut_refs, unreachable_code)]
    pub fn join(self) -> T {
        unsafe { SCHED.block_on_join(self.child); }
        self.result.take()
    }
}

// ================================
// Scheduler skeleton (iretq path)
// ================================

use alloc::collections::BTreeMap;
use crate::lib_alloc::is_heap_addr;
use crate::{hlt_loop, serial_println};

pub struct Scheduler {
    pub runq: VecDeque<ThreadId>,
    pub tcb: BTreeMap<ThreadId, Thread>,
    pub current_id: Option<ThreadId>,
    pub next_id: ThreadId,
}

impl Default for Scheduler {
    fn default() -> Self {
        Self::new()
    }
}

impl Scheduler {
    pub const fn new() -> Self {
        Self {
            runq: VecDeque::new(),
            tcb: BTreeMap::new(),
            current_id: None,
            next_id: 1,
        }
    }

    #[allow(static_mut_refs)]
    pub fn spawn<F, T>(&mut self, f: F, stack_size: usize) -> JoinHandle<T>
    where
        F: FnOnce() -> T + Send + 'static,
        T: Send + 'static,
    {
        // Create the join result slot and an erased pointer for the TCB
        let result = Box::new(ThreadResult::<T>::new());
        let result_erased = NonNullErasedResult::new(result.as_ref());

        // Spawn context stored on the new thread's stack
        struct SpawnCtx<U: Send + 'static> {
            func: Option<Box<dyn FnOnce() -> U + Send + 'static>>,
            result_ref: NonNullErasedResult,
        }

        extern "C" fn start_with_ctx<U>(ctx_ptr: *mut ()) -> !
        where
            U: Send + 'static,
        {
            serial_println!("IS ({:?} A HEAP ADDR? {})", ctx_ptr, is_heap_addr(ctx_ptr as usize));
            unsafe {
                let ctx = &mut *(ctx_ptr as *mut SpawnCtx<U>);
                let v = (ctx.func.take().expect("spawn func missing"))();
                ctx.result_ref.as_ref::<U>().set(v);
                SCHED.finish_current_and_wake_joiners();
            }
        }

        let mut ctx = Box::new(SpawnCtx::<T> {
            func: Some(Box::new(f)),
            result_ref: result_erased,
        });

        let id = self.alloc_id();

        // Create the Thread (this calls init_stack_frame which returns a pointer pointing at RIP)
        let mut thread = Thread::new(id, stack_size, start_with_ctx::<T>);
        thread.join_raw = Some(result_erased);

        // Fill the ctx slot in the stack layout (matches init_stack_frame with PAD)
        //
        // Layout (addresses increasing):
        //   [RIP]        <- thread.stack_ptr (returned by init_stack_frame)
        //   [CS]
        //   [RFLAGS]
        //   [PAD]        // fixed 8 bytes inserted by init_stack_frame
        //   [ctx]        <- ctx_slot (we write ctx here)
        //   [start_fn]   <- start_with_ctx address (written by init_stack_frame)
        //   [callee-saved placeholders...]
        unsafe {
            use core::mem::size_of;
            let q = size_of::<u64>();
            let frame_words = 3 * q; // RIP, CS, RFLAGS
            let pad = q;             // single 8-byte PAD inserted by init_stack_frame

            // ctx is at stack_ptr + frame_words + pad
            let ctx_slot = (thread.stack_ptr as usize + frame_words + pad) as *mut u64;
            // start_fn is at stack_ptr + frame_words + pad + q
            let start_slot = (thread.stack_ptr as usize + frame_words + pad + q) as *const u64;

            // Write ctx pointer into the ctx slot
            *ctx_slot = (&mut *ctx as *mut SpawnCtx<T>) as u64;

            // Read back and sanity-check the start_fn and ctx we expect
            let start_val = *start_slot;
            let ctx_val = *ctx_slot;
            serial_println!("SPAWN CHECK: thread id={} start={:#018x} ctx={:#018x}", id, start_val, ctx_val);

            debug_assert_eq!(start_val, start_with_ctx::<T> as *const () as u64, "start_fn mismatch in stack layout");
            debug_assert!(ctx_val != 0, "ctx pointer is null");

            // Bounds check: ensure stack_ptr lies within the allocated buffer
            let base = thread._stack_mem.as_ptr() as usize;
            let top = base + thread._stack_mem.len();
            debug_assert!(
                (thread.stack_ptr as usize) >= base && (thread.stack_ptr as usize) < top,
                "stack_ptr out of bounds"
            );
        }

        serial_println!("spawned thread (id {}).", thread.id);
        self.tcb.insert(id, thread);
        self.runq.push_back(id);

        JoinHandle { child: id, result }
    }

    /// # Safety
    /// must be valid.
    #[allow(static_mut_refs)]
    pub unsafe fn start(&mut self) -> ! {
        serial_println!("starting threads.");
        let next = self.runq.pop_front().expect("no threads to start");
        self.current_id = Some(next);

        static mut BOOT_SP_SAVE: *mut u8 = core::ptr::null_mut();
        let old_sp_ptr: *mut *mut u8 = unsafe { &mut BOOT_SP_SAVE };
        let new_sp_ptr: *const *mut u8 = &self.tcb.get(&next).unwrap().stack_ptr;

        serial_println!("switching stack via iretq.");
        // Keep interrupts disabled here; iretq will restore IF from the thread frame.
        unsafe { 
            dump_and_assert_next_frame(new_sp_ptr);
            switch_interrupt_context_raw(old_sp_ptr, new_sp_ptr);
            unreachable_unchecked();
        }
    }

    pub unsafe fn block_on_join(&mut self, child: ThreadId) -> ! {
        let me = self.current_id.expect("no current thread");
        let child_tcb = self.tcb.get_mut(&child).expect("invalid child");
        child_tcb.waiters.push(me);

        {
            let me_tcb = self.tcb.get_mut(&me).unwrap();
            me_tcb.state = ThreadState::Blocked(BlockReason::Join { child });
        }

        let next = self.runq.pop_front().expect("no runnable threads");
        self.current_id = Some(next);

        let old_sp_ptr: *mut *mut u8 = &mut self.tcb.get_mut(&me).unwrap().stack_ptr;
        let new_sp_ptr: *const *mut u8 = &self.tcb.get(&next).unwrap().stack_ptr;

        unsafe { 
            switch_interrupt_context_raw(old_sp_ptr, new_sp_ptr);
            unreachable_unchecked();
        }
    }

    pub unsafe fn finish_current_and_wake_joiners(&mut self) -> ! {
        let me = self.current_id.expect("no current thread");
        let (waiters, next_sp_ptr) = {
            let me_tcb = self.tcb.get_mut(&me).unwrap();
            me_tcb.state = ThreadState::Terminated;
            let ws = core::mem::take(&mut me_tcb.waiters);
            let next = self.runq.pop_front().unwrap_or_else(|| self.spawn_idle_if_needed());
            self.current_id = Some(next);
            let next_sp_ptr = &self.tcb.get(&next).unwrap().stack_ptr as *const *mut u8;
            (ws, next_sp_ptr)
        };

        for w in waiters {
            if let Some(t) = self.tcb.get_mut(&w) {
                if matches!(t.state, ThreadState::Blocked(BlockReason::Join { child: c }) if c == me)
                {
                    t.state = ThreadState::Ready;
                    self.runq.push_back(w);
                }
            }
        }

        let old_sp_ptr: *mut *mut u8 = &mut self.tcb.get_mut(&me).unwrap().stack_ptr;
        unsafe { 
            switch_interrupt_context_raw(old_sp_ptr, next_sp_ptr); 
            unreachable_unchecked()
        }
    }

    fn alloc_id(&mut self) -> ThreadId {
        let id = self.next_id;
        self.next_id += 1;
        id
    }

    fn spawn_idle_if_needed(&mut self) -> ThreadId {
        extern "C" fn idle(_: *mut ()) -> ! {
            hlt_loop()
        }
        let id = self.alloc_id();
        let idle = Thread::new(id, 16 * 1024, idle);
        self.tcb.insert(id, idle);
        id
    }

    // Cooperative yield implemented via iretq path as well (keeps single switch primitive)
    pub unsafe fn cooperative_yield(&mut self) -> ! {
        let me = self.current_id.expect("no current thread");
        let next = self.pick_next_ready(me);

        {
            let me_tcb = self.tcb.get_mut(&me).unwrap();
            me_tcb.state = ThreadState::Ready;
            self.runq.push_back(me);
        }

        self.current_id = Some(next);

        let old_sp_ptr: *mut *mut u8 = &mut self.tcb.get_mut(&me).unwrap().stack_ptr;
        let new_sp_ptr: *const *mut u8 = &self.tcb.get(&next).unwrap().stack_ptr;

        unsafe { 
            switch_interrupt_context_raw(old_sp_ptr, new_sp_ptr);
            unreachable_unchecked()
        }
    }

    // Preemption from timer: save current, pick next, switch (called from ISR)
    pub unsafe fn preempt_current(&mut self) -> ! {
        let Some(me) = self.current_id else { unsafe { hint::unreachable_unchecked() } };

        let next = self.pick_next_ready(me);
        if next == me {
            // No other thread to run; return from ISR normally (caller should return)
            unsafe { hint::unreachable_unchecked() };
        }

        {
            let me_tcb = self.tcb.get_mut(&me).unwrap();
            if matches!(me_tcb.state, ThreadState::Running) {
                me_tcb.state = ThreadState::Ready;
                self.runq.push_back(me);
            }
        }

        self.current_id = Some(next);

        let old_sp_ptr: *mut *mut u8 = &mut self.tcb.get_mut(&me).unwrap().stack_ptr;
        let new_sp_ptr: *const *mut u8 = &self.tcb.get(&next).unwrap().stack_ptr;

        // In ISR, interrupts are disabled by hardware entry. iretq will restore IF.
        unsafe { 
            switch_interrupt_context_raw(old_sp_ptr, new_sp_ptr);
            unreachable_unchecked()
        }
    }

    pub fn pick_next_ready(&mut self, current: ThreadId) -> ThreadId {
        for _ in 0..self.runq.len() {
            if let Some(id) = self.runq.pop_front() {
                if id != current {
                    if matches!(self.tcb.get(&id).unwrap().state, ThreadState::Ready) {
                        return id;
                    } else {
                        self.runq.push_back(id);
                    }
                } else {
                    self.runq.push_back(id);
                }
            }
        }
        current
    }
}

// ================================
// Public top-level API

#[allow(static_mut_refs)]
pub mod rt {
    use alloc::{boxed::Box, vec::Vec};

    use super::SCHED;
    use crate::thread::{Thread, ThreadState};

    /// Initialize the runtime scheduler.
    /// Must be called once at kernel entry before enabling interrupts.
    pub unsafe fn boot() {
        let id = 0;

        let bootstrap = Thread {
            id,
            state: ThreadState::Running,
            _stack_mem: Box::new([]),
            stack_ptr: core::ptr::null_mut(),
            waiters: Vec::new(),
            join_raw: None,
        };

        unsafe {
            SCHED.tcb.insert(id, bootstrap);
            SCHED.current_id = Some(id);
            SCHED.runq.clear();
        }
    }

    /// Start running the first thread
    pub unsafe fn start() -> ! {
        unsafe { SCHED.start() }
    }

    /// Cooperative yield from running thread
    pub unsafe fn yield_now() -> ! {
        unsafe { SCHED.cooperative_yield() }
    }

    /// Timer IRQ handler: preemptive scheduling
    /// Must be invoked from your ISR after acknowledging the IRQ.
    pub unsafe fn on_timer_tick() {
        // Call SCHED.preempt_current() from ISR context.
        unsafe { SCHED.preempt_current() };
    }
}

pub static mut SCHED: Scheduler = Scheduler::new();

pub static INIT: AtomicBool = AtomicBool::new(false);
